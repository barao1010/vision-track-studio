cmake_minimum_required(VERSION 3.20)
project(VisionIndependentCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(vision_core src/core.cpp src/media.cpp)
target_include_directories(vision_core PUBLIC src)
if(MSVC)
  target_compile_options(vision_core PRIVATE /W4 /WX)
else()
  target_compile_options(vision_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE vision_core)
enable_testing()
add_test(NAME core_contracts COMMAND core_tests)

add_executable(render_fixture validation/render_fixture.cpp)
target_link_libraries(render_fixture PRIVATE vision_core)
add_executable(media_tests tests/media_tests.cpp)
target_link_libraries(media_tests PRIVATE vision_core)
add_test(NAME media_contracts COMMAND media_tests ${CMAKE_CURRENT_BINARY_DIR}/media-test-output)

if(WIN32)
  add_executable(vision_studio WIN32 app/windows.cpp)
  target_link_libraries(vision_studio PRIVATE vision_core windowscodecs ole32 comdlg32 gdi32 user32)
  add_test(NAME windows_gui_actions COMMAND vision_studio --self-test-gui)
  set_tests_properties(windows_gui_actions PROPERTIES TIMEOUT 30)
endif()
