cmake_minimum_required(VERSION 3.20)
project(VisionIndependentCore LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(vision_core src/core.cpp src/media.cpp src/audio_rate.cpp src/audio_buffer.cpp)
target_include_directories(vision_core PUBLIC src)
if(MSVC)
  target_compile_options(vision_core PRIVATE /W4 /WX)
else()
  target_compile_options(vision_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE vision_core)
enable_testing()
add_test(NAME core_contracts COMMAND core_tests)

add_executable(render_fixture validation/render_fixture.cpp)
target_link_libraries(render_fixture PRIVATE vision_core)
add_executable(media_tests tests/media_tests.cpp)
target_link_libraries(media_tests PRIVATE vision_core)
add_test(NAME media_contracts COMMAND media_tests ${CMAKE_CURRENT_BINARY_DIR}/media-test-output)

if(WIN32)
  add_library(vision_windows_desktop platform/windows_desktop.cpp)
  target_include_directories(vision_windows_desktop PUBLIC platform)
  target_link_libraries(vision_windows_desktop PUBLIC vision_core gdi32 user32)
  add_executable(vision_studio WIN32 app/windows.cpp)
  target_link_libraries(vision_studio PRIVATE vision_windows_recording vision_windows_desktop windowscodecs ole32 comdlg32 gdi32 user32)
  add_test(NAME windows_gui_actions COMMAND vision_studio --self-test-gui)
  set_tests_properties(windows_gui_actions PROPERTIES TIMEOUT 30)
endif()

if(WIN32)
  add_library(vision_windows_recording platform/windows_recording.cpp)
  target_include_directories(vision_windows_recording PUBLIC platform)
  target_link_libraries(vision_windows_recording PUBLIC vision_core mfplat mfreadwrite mfuuid ole32)
  add_executable(windows_recording_test tests/windows_recording_test.cpp)
  target_link_libraries(windows_recording_test PRIVATE vision_windows_recording)
  add_test(NAME windows_native_recording COMMAND windows_recording_test ${CMAKE_CURRENT_BINARY_DIR}/native-recording.mp4)
  set_tests_properties(windows_native_recording PROPERTIES TIMEOUT 60)
endif()
if(WIN32)
  add_executable(windows_desktop_test tests/windows_desktop_test.cpp)
  target_link_libraries(windows_desktop_test PRIVATE vision_windows_desktop)
  add_test(NAME windows_desktop_capture COMMAND windows_desktop_test)
  set_tests_properties(windows_desktop_capture windows_gui_actions PROPERTIES RUN_SERIAL TRUE TIMEOUT 30)
endif()

add_executable(audio_rate_tests tests/audio_rate_tests.cpp)
target_link_libraries(audio_rate_tests PRIVATE vision_core)
add_test(NAME audio_rate_contracts COMMAND audio_rate_tests)

add_executable(audio_buffer_tests tests/audio_buffer_tests.cpp)

add_test(NAME audio_buffer_contracts COMMAND audio_buffer_tests)

  target_include_directories(vision_windows_microphone PUBLIC platform)
  target_link_libraries(vision_windows_microphone PUBLIC vision_core ole32 uuid)
  target_link_libraries(vision_studio PRIVATE vision_windows_microphone)
endif()
